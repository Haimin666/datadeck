## 发送逻辑: 
mysql 表 sqgr_dz_con_sms_send

| sqgr_dz_con_sms_send | CREATE TABLE `sqgr_dz_con_sms_send` (
 **`pay_sch_no` varchar(50) NOT NULL COMMENT '支付表编号(去重键)',**
** `pay_sch_id` bigint(20) NOT NULL DEFAULT '0' COMMENT '支付表id',**
** `cst_id` bigint(20) NOT NULL DEFAULT '0' COMMENT '客户id',**
** `cst_nm` varchar(100) NOT NULL DEFAULT '' COMMENT '客户名称',**
** `mobile` varchar(20) NOT NULL DEFAULT '' COMMENT '手机号',**
** `prd_sts_cd` varchar(20) NOT NULL DEFAULT '' COMMENT '资产状态码',**
** `stl_dt` date NOT NULL DEFAULT '1970-01-01' COMMENT '结清时间',**
** `fee_amt` decimal(16,2) NOT NULL DEFAULT '0.00' COMMENT '本金费用合计',**
** `ar_amt` decimal(16,2) NOT NULL DEFAULT '0.00' COMMENT '未核销金额(剩余本金)',**
** `vf_amt` decimal(16,2) NOT NULL DEFAULT '0.00' COMMENT '已核销金额',**
** `remain_amt` decimal(16,2) NOT NULL DEFAULT '0.00' COMMENT '剩余本金',**
** `remind_type` varchar(20) NOT NULL DEFAULT '' COMMENT '提醒类型 NEW=新增 MONTHLY=历史',**
** `intime` date NOT NULL DEFAULT '1970-01-01' COMMENT '提醒生成时间',**
 `is_send_success` tinyint(1) NOT NULL DEFAULT '0' COMMENT '是否发送成功 (1:成功, 0:失败)',
 `create_time` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '数据同步入库时间',
 PRIMARY KEY (**`pay_sch_no`,`remind_type`,`intime**`),
 KEY `idx_cst_id` (`cst_id`) COMMENT '客户维度查询索引'
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='狮桥个人-呆账短信提醒表' |

怎么保证每次发送的数据都是有效的 :
通过联合主键**pay_sch_no ,remind_type,intime  ,通过  "writeMode": "update", 更新数据,datax 同步的时候没同步is_send_success 这个字段,都是新增数据的时候默认是未发送 ,短信成功发送后is_send_success会被标记1 ,对于相同的联合主键程序重新运行后不会影响发送状态is_send_success**
发送规则是 只取 is_send_success = 0 AND intime <= 当天。未到期的存量不会提前发。
