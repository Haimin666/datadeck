**【需求背景】：**
        小蜜订单8-10%发生逾期，因为目前只上报借款人征信，没有其他的措施，如果借款人资金链出现问题，无还款能力时，也很难找到其他的资产或者亲属替借款人还钱；
坏账率预估是2.13%，基本逾期15+的可以通过电催和销售催回，30+以上基本无催回。
小蜜业务提出小蜜花业务客户催收困难，期望增加反担保人征信，来降低催收难度。
 
**【需求内容】：**
 人行征信上报反担保人信息，增加催收渠道

【**征信开发】：**
同步赵冉新增的担保人信息表：terminal_finance.access_credit_report_record
限制条件：limit_biz_type=1           --0小贷;1小蜜
                  sign_status=1               --0签署;1签署

guarantor_cert_no   --担保人身份证号

insert into  TP_CRS_GUAR_ACCT_BS_INFSGMT
 (CUSTID,
 DATA_SOURCE,
 --FINORGCODE,
 ACCTTYPE,
 ACCTCODE,
 --MNGMTORGCODE,
 BUSILINES,
 BUSIDTILLINES,
 OPENDATE,
 DUEDATE,
 CLOSEDATE,
 ACCTSTATUS,
 ACCTCREDLINE,
 CY,
 CTRCTTXTCD,
 GUARMODE,
 OTHREPYGUARWAY,
 SECDEP,
 COMPADVDATE,
 COMPADVFLAG)
 SELECT 
 TB.CR_CNT_NO ACCTCODE, ----账户标识号码
 '12000', ----DATA_SOURCE
 '1' InfoIDType, --身份类别, 1:自然人，2：组织机构
 fdb.guarantor_user_name ArlpName, --责任人名称
 '1' ArlpCertType, --责任人身份标识类型
 fdb.guarantor_cert_no ArlpCertNum, --责任人身份标识号码
 '2' ArlpType, --还款责任人类型: 2-反担保人
 RUNC(TB.CR_AMT / 100) ArlpAmt, --还款责任金额
 '0' Wartysign, --联保标志
 TB.CR_CNT_NO MaxGuarMcc --保证合同标识码
 
 FROM (SELECT PR.*
 FROM FLUSR.MCR_BILL_M@LION_LINK PR --（小贷账单主表）
 WHERE PR.IS_DEL = 0 --未删除
 --AND (TRUNC(PR.CR_DT) BETWEEN TRUNC(B_DATE) AND TRUNC(E_DATE)) --起租日--小贷的放款日
 AND PR.LOAN_STS_CD IN ('4', '6', '7', '8') --已放款的数据
 AND PR.LOAN_BIZ_TYP_CD <> '3' --排除垫款贷
 AND PR.CR_DT < trunc(v_sysdate)
 AND REP_STS IN ('09','00','03')
 AND REPORT_STATUS=1
 AND CST_TYPE_CD in ('0','1','2','3','4','5','6','7')
 --AND T.IS_UPLOAD_CREDIT=1--我司上报标志，只在起租信息加，保证已报送的继续报送。
 --=======================================================================================
 ) TB
 LEFT JOIN (SELECT BILL_ID, MAX(T.PAY_DT) PAY_DT
 FROM FLUSR.MCR_BILL_D@LION_LINK T --（小贷账单明细表）
 WHERE IS_DEL = 0
 GROUP BY T.BILL_ID) PD
 ON TB.ID = PD.BILL_ID
 left join (select PRJ_ID from flusr.mcr_inform_transfer@LION_LINK 
 where is_del=0 
 and REPO_SUBJECT=1
 --wh 20200420
 --and REPO_SUBJECT IN (V_REPO_SUBJECT) --小贷回购主体1.卓远，2.狮桥
 ) mt on mt.PRJ_ID=[tb.id](http://tb.id)
 INNER JOIN (SELECT T.ID, T.CERT_TYP_CD, T.CERT_NO, T.CST_NM
 FROM FLUSR.CST_BSC_INFO@LION_LINK T --（自然人客户基础信息）
 WHERE T.CST_TYP_CD = 1 --自然人
 AND T.IS_DEL = 0) CB
 ON TB.CST_ID = CB.ID
 left join 
 (
 --反担保人查询
 select 
 guarantor_user_name --担保人姓名
 ,guarantor_cert_no --担保人身份证号码
 from terminal_finance.access_credit_report_record
 where limit_biz_type=1 --0小贷;1小蜜
 and sign_status=1 --0签署;1签署
 ) fdb   
 LEFT JOIN (SELECT CONT FROM ZY_P_CONT_JQDATA WHERE TYPE = 'DB') ZJ
 ON TB.CR_CNT_NO = ZJ.CONT
 WHERE ZJ.CONT IS NULL and mt.PRJ_ID is null;
